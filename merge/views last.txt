<script>
let currentPage = 1;
const rowsPerPage = 10;
let currentSort = { column: null, ascending: true };

function applyFilters() {
  const selectedUsernames = Array.from(document.getElementById('usernameFilter').selectedOptions).map(opt => opt.value);
  const selectedNotes = Array.from(document.getElementById('lognoteFilter').selectedOptions).map(opt => opt.value);
  const startDate = document.getElementById('startDate').value;
  const endDate = document.getElementById('endDate').value;

  const rows = document.querySelectorAll('#logsBody tr');
  rows.forEach(row => {
    const username = row.dataset.username;
    const action = row.dataset.action;
    const timestamp = row.dataset.timestamp;
    const date = new Date(timestamp).toISOString().split('T')[0];

    const matchUser = selectedUsernames.length === 0 || selectedUsernames.includes(username);
    const matchNote = selectedNotes.length === 0 || selectedNotes.includes(action);
    const matchStart = !startDate || date >= startDate;
    const matchEnd = !endDate || date <= endDate;

    row.classList.toggle('filtered-in', matchUser && matchNote && matchStart && matchEnd);
  });

  currentPage = 1;
  renderTable();
  renderPagination();
}

function resetFilters() {
  document.getElementById('filterForm').reset();
  Array.from(document.querySelectorAll('#filterForm select')).forEach(select => select.selectedIndex = -1);
  document.querySelectorAll('#logsBody tr').forEach(row => row.classList.add('filtered-in'));
  currentPage = 1;
  renderTable();
  renderPagination();
}

function renderTable() {
  const rows = Array.from(document.querySelectorAll('#logsBody tr'));
  const filteredRows = rows.filter(row => row.classList.contains('filtered-in'));
  const start = (currentPage - 1) * rowsPerPage;
  const end = start + rowsPerPage;

  rows.forEach(row => row.style.display = 'none');

  filteredRows.forEach((row, index) => {
    if (index >= start && index < end) {
      row.style.display = '';
      row.children[0].textContent = index + 1;
    }
  });

  if (filteredRows.length === 0) {
    document.getElementById('logsBody').innerHTML = '<tr><td colspan="5" class="text-center text-muted">No logs found for the selected filters.</td></tr>';
  }
}

function renderPagination() {
  const rows = Array.from(document.querySelectorAll('#logsBody tr.filtered-in'));
  const totalPages = Math.ceil(rows.length / rowsPerPage);
  const container = document.getElementById('pagination');
  container.innerHTML = '';

  for (let i = 1; i <= totalPages; i++) {
    const btn = document.createElement('button');
    btn.textContent = i;
    btn.className = 'btn btn-sm btn-outline-primary me-1';
    if (i === currentPage) btn.classList.add('active');
    btn.addEventListener('click', () => {
      currentPage = i;
      renderTable();
      renderPagination();
    });
    container.appendChild(btn);
  }
}

function sortByColumn(column) {
  const tbody = document.getElementById('logsBody');
  const rows = Array.from(tbody.querySelectorAll('tr.filtered-in'));
  const ascending = currentSort.column === column ? !currentSort.ascending : true;
  currentSort = { column, ascending };

  rows.sort((a, b) => {
    let valA = a.dataset[column];
    let valB = b.dataset[column];
    if (column === 'timestamp') {
      valA = new Date(valA);
      valB = new Date(valB);
    }
    return ascending ? (valA > valB ? 1 : -1) : (valA < valB ? 1 : -1);
  });

  rows.forEach(row => tbody.appendChild(row));
  currentPage = 1;
  renderTable();
  renderPagination();
}

function exportToCSV() {
  const rows = Array.from(document.querySelectorAll('#logsBody tr.filtered-in')).filter(row => row.style.display !== 'none');
  let csv = 'Username,Action,IP Address,Timestamp\n';
  rows.forEach(row => {
    const cols = row.querySelectorAll('td');
    csv += Array.from(cols).slice(1).map(td => `"${td.textContent.trim()}"`).join(',') + '\n';
  });
  const blob = new Blob([csv], { type: 'text/csv' });
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = 'access_logs.csv';
  a.click();
  URL.revokeObjectURL(url);
}

function exportToExcel() {
  const table = document.getElementById('logsTable');
  const rows = Array.from(document.querySelectorAll('#logsBody tr.filtered-in')).filter(row => row.style.display !== 'none');
  const html = '<table>' + table.querySelector('thead').outerHTML + '<tbody>' + rows.map(row => row.outerHTML).join('') + '</tbody></table>';
  const blob = new Blob([html], { type: 'application/vnd.ms-excel' });
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = 'access_logs.xls';
  a.click();
  URL.revokeObjectURL(url);
}

document.addEventListener('DOMContentLoaded', () => {
  document.querySelectorAll('#logsBody tr').forEach(row => row.classList.add('filtered-in'));
  renderTable();
  renderPagination();
});
</script>

                serverTimeDisplay.textContent = serverTime.toLocaleString('fr-FR');

                console.error('Error fetching server time:', error);
                serverTimeDisplay.textContent = 'Échec du chargement';
            }
            
            const localTime = new Date();
            localTimeDisplay.textContent = localTime.toLocaleString('fr-FR');
            localTimeInput.value = localTime.toISOString();
        };

        await updateTimes();
        setInterval(updateTimes, 30000);

        usernameInput.addEventListener('change', async (event) => {
            const username = event.target.value.trim();
            if (username.length > 0) {
                try {
                    const csrfToken = document.getElementById('csrfToken').value;
                    const response = await fetch(window.location.origin + '/auth/get-user-facility-and-role', {
                        method: 'POST',
                        headers: {
                            'Content-Type': 'application/json',
                            'x-csrf-token': csrfToken
                        },
                        body: JSON.stringify({ username }),
                    });
                    const data = await response.json();
                    
                    if (data.success) {
                        facilityNameInput.value = data.facilityName;
                        userRoleNameInput.value = data.roleName;
                    } else {
                        facilityNameInput.value = data.message;
                        userRoleNameInput.value = '';
                    }
                } catch (error) {
                    console.error('Error fetching user info:', error);
                    facilityNameInput.value = 'Erreur lors du chargement des informations de l\'utilisateur.';
                    userRoleNameInput.value = '';
                }
            } else {
                facilityNameInput.value = 'Saisir le nom d\'utilisateur pour charger l\'établissement';
                userRoleNameInput.value = 'Saisir le nom d\'utilisateur pour charger le rôle';
            }
        });

        if (usernameInput.value) {
            usernameInput.dispatchEvent(new Event('change'));
        }
    });
</script>

        const facilityTypeSelect = document.getElementById('facility-type');

        const parentIdSelect = parentFacilityField.querySelector('select[name="parent_id"]');

        if (facilityTypeSelect) {

            if (isSatellite === 'true') {
                parentFacilityField.style.display = 'block';
            } else {
                parentFacilityField.style.display = 'none';
            }

                if (facilityTypeSelect.value === 'satellite') {
                    parentFacilityField.style.display = 'block';
                    parentIdSelect.setAttribute('required', 'required');
                } else {
                    parentFacilityField.style.display = 'none';
                    parentIdSelect.removeAttribute('required');
                }
            };

            facilityTypeSelect.addEventListener('change', toggleParentField);
        }
    });

<script>
  // Debounce function to limit API calls
  function debounce(func, delay) {
    let timeout;
    return function(...args) {
      const context = this;
      clearTimeout(timeout);
      timeout = setTimeout(() => func.apply(context, args), delay);
    };
  }

  // Dynamic personnel search
  document.getElementById('personSearch').addEventListener('input', debounce(function () {
    const searchQuery = this.value;
    const selectElement = document.getElementById('person_id');

    // Clear previous search results, but keep the default option
    while (selectElement.options.length > 1) {
      selectElement.remove(1);
    }

    if (searchQuery.length < 3) {
      return;
    }

    fetch(`/admin/users/personnel-search?query=${encodeURIComponent(searchQuery)}`)
      .then(response => {
        if (!response.ok) {
          throw new Error('Network response was not ok');
        }
        return response.json();
      })
      .then(data => {
        data.forEach(person => {
          const option = document.createElement('option');
          option.value = person.pid;
          option.textContent = `${person.name_first || ''} ${person.name_last || ''} (PID: ${person.pid})`;
          option.dataset.firstname = person.name_first || '';
          option.dataset.lastname = person.name_last || '';
          option.dataset.email = person.email || '';
          selectElement.appendChild(option);
        });
      })
      .catch(error => {
        console.error('Error fetching search results:', error);
        const errorOption = document.createElement('option');
        errorOption.textContent = 'Failed to load personnel. Please try again.';
        errorOption.disabled = true;
        selectElement.appendChild(errorOption);
      });
  }, 300));

  // Dynamic username and email generation on selection
  document.getElementById('person_id').addEventListener('change', function () {
    const selectedOption = this.options[this.selectedIndex];
    const usernameField = document.getElementById('username');
    const emailField = document.getElementById('email');

    if (selectedOption.value) {
      const firstName = selectedOption.dataset.firstname;
      const lastName = selectedOption.dataset.lastname;
      // const pid = selectedOption.value; // Removed pid
      const email = selectedOption.dataset.email;

      let generatedUsername = '';
      if (lastName) {
        generatedUsername += lastName.toLowerCase().replace(/[^a-z0-9]/g, '');
      }
      if (firstName && firstName.length >= 2) {
        generatedUsername += firstName.substring(0, 2).toLowerCase().replace(/[^a-z0-9]/g, '');
      } else if (firstName) {
        generatedUsername += firstName.toLowerCase().replace(/[^a-z0-9]/g, '');
      }
      // generatedUsername += pid; // REMOVED: No longer appending pid

      usernameField.value = generatedUsername;
      emailField.value = email;
    } else {
      usernameField.value = '';
      emailField.value = '';
    }
  });
</script>
