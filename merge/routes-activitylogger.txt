const express = require('express');
const router = express.Router();
const { Op } = require('sequelize');
const { care_accesslog } = require('../models');
const authMiddleware = require('../middleware/authMiddleware');
const isAdmin = require('../middleware/isAdminMiddleware');

// GET: /admin/logs/
router.get('/', authMiddleware, isAdmin, async (req, res) => {
  const { username, from, to, success } = req.query;

  const where = {};

  if (username) {
    where.username = { [Op.like]: `%${username}%` };
  }

  if (from || to) {
    where.datetime = {};
    if (from) where.datetime[Op.gte] = new Date(from);
    if (to) where.datetime[Op.lte] = new Date(to);
  }

  if (success === '1' || success === '0') {
    where.login_success = success === '1';
  }

  try {
    const logs = await care_accesslog.findAll({
      where,
      order: [['datetime', 'DESC']],
      limit: 100,
    });

    res.render('admin/logs', {
      title: 'Access Logs',
      logs,
      filters: { username, from, to, success },
      activePage: 'admin-logs',
      user: req.user,
    });
  } catch (error) {
    console.error('Error fetching logs:', error);
    res.status(500).send('Failed to load access logs.');
  }
});

module.exports = router;

// routes/updateUser.js
const express = require('express');
const router = express.Router();
const userController = require('../controllers/userController'); // NEW: Import the user controller
const userUpdateValidator = require('../middleware/userUpdateValidator'); // NEW: Import the validator
const { User, care_user_roles, care_department, care_staff, care_person, care_facilities } = require('../models');

// GET: Show update form with user selection dropdown
router.get('/', async (req, res) => {
  try {
    const users = await User.findAll({
      order: [['username', 'ASC']],
      include: [
        { model: care_user_roles, as: 'userRole' },
        { model: care_staff, as: 'staff', include: [{ model: care_person, as: 'person' }] },
        { model: care_department, as: 'department' }
      ]
    });

    const roles = await care_user_roles.findAll({ order: [['role_name', 'ASC']] });
    const departments = await care_department.findAll({ where: { is_inactive: false }, order: [['name_formal', 'ASC']] });
    const facilities = await care_facilities.findAll({ order: [['name', 'ASC']] });

    const selectedUserId = req.query.user_id;
    let userToEdit = null;
    if (selectedUserId) {
      userToEdit = await User.findByPk(selectedUserId, {
        include: [
          { model: care_user_roles, as: 'userRole' },
          { model: care_staff, as: 'staff', include: [{ model: care_person, as: 'person' }] },
          { model: care_department, as: 'department' },
          { model: care_facilities, as: 'facility' }
        ]
      });
    }

    res.render('users/updateUser', {
      title: 'Update Account',
      users,
      userToEdit,
      roles,
      departments,
      facilities,
      user: req.user,
      success: req.query.success,
      errors: [],
      activePage: 'admin-update-user'
    });
  } catch (error) {
    console.error('Error loading update form:', error);
    res.status(500).send('Failed to load update form.');
  }
});

// POST: Handle user update
// UPDATED: Now uses the validator middleware and a dedicated controller
router.post('/edit', userUpdateValidator, userController.updateUser);

module.exports = router;

// utils/activityLogger.js
const { care_accesslog } = require('../models');

/**
 * Logs an activity to the care_accesslog table.
 * @param {object} req - The Express request object.
 * @param {string} lognote - A descriptive note of the activity.
 * @param {boolean} login_success - True for success, false for failure.
 * @param {string} thisfile - The file/route where the activity occurred.
 * @param {number|string} [userId] - The user ID to explicitly log (optional).
 * @param {string} [username] - The username to explicitly log (optional).
 */
const logActivity = async (req, lognote, login_success, thisfile, userId, username) => {
    try {
        // Use the provided userId and username, or fall back to req.user if they are not provided.
        const userIdentifier = userId || (req.user ? req.user.user_id : 'unknown');
        const userDisplayName = username || (req.user ? req.user.username : 'system');
        const ip = req.ip || req.headers['x-forwarded-for'] || req.connection.remoteAddress;

        await care_accesslog.create({
            datetime: new Date(),
            ip,
            lognote,
            userid: String(userIdentifier), // Ensure it's a string for the DB
            username: userDisplayName,
            thisfile,
            fileforward: req.originalUrl,
            login_success: login_success ? 1 : 0,
        });
    } catch (error) {
        console.error('Failed to log activity:', error);
    }
};

module.exports = logActivity;
